    description = models.CharField(max_length=2000,blank=True, null=True)
    file_img_video = FileField(upload_to='ads/')

    def __str__(self):
        return self.title

    title = models.CharField(max_length=250,blank=True, null=True)
    description =  models.CharField(max_length=2000,blank=True, null=True)
    image = FileField(upload_to='banners/')

    def __str__(self):
        return self.title

class Category(models.Model):
    title = models.CharField(max_length=300, unique=True, default='default_title')
    image = models.FileField(upload_to='category_images/')

    def __str__(self):
        return self.title

    title = models.CharField(max_length=300, unique=True,default='default_title')
    image = models.FileField(upload_to='subcategory_images/')

    def __str__(self):
        return self.title

class CatVideosSerializer(serializers.ModelSerializer):
    videos = serializers.SerializerMethodField()

    def get_videos(self, obj):
        request = self.context.get('request')
        return request.build_absolute_uri(obj.videos.url) if obj.videos else None

    class Meta:
        model = CatVideos
        fields = ['id', 'title','videos','created_at','updated_at']


class SubCatVideosSerializer(serializers.ModelSerializer):
    videos = serializers.SerializerMethodField()

    def get_videos(self, obj):

    class Meta:
        model = SubCatVideos
        fields = ['id', 'title','videos','created_at','updated_at']

class CategoriesScreen extends StatefulWidget {
  const CategoriesScreen({super.key});

  @override
  _CategoriesScreenState createState() => _CategoriesScreenState();
}

class _CategoriesScreenState extends State<CategoriesScreen> {
  List categories = [];
  bool isLoading = true;

  @override
  void initState() {
    super.initState();
    fetchCategories();
  }

  Future<void> fetchCategories() async {
    final response = await http
        .get(Uri.parse('http://192.168.192.133:8000/api/categories/'));
    if (response.statusCode == 200) {
      setState(() {
        categories = json.decode(response.body);
        isLoading = false;
      });
    } else {
      throw Exception('Failed to load categories');
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Categories')),
      body: isLoading
          ? const Center(child: CircularProgressIndicator())
          : GridView.builder(
              gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
                crossAxisCount: 2,
                childAspectRatio: 3 / 2,
              ),
              itemCount: categories.length,
              itemBuilder: (context, index) {
                var category = categories[index];
                return GestureDetector(
                  onTap: () {
                    if (category['videos'].isEmpty) {
                      Navigator.push(
                        context,
                        MaterialPageRoute(
                          builder: (context) => SubcategoriesScreen(
                            subcategories: category['subcategories'],
                          ),
                        ),
                      );
                    } else {
                      Navigator.push(
                        context,
                        MaterialPageRoute(
                          builder: (context) =>
                              VideosScreen(videos: category['videos']),
                        ),
                      );
                    }
                  },
                  child: Card(
                    child: Column(
                      mainAxisAlignment: MainAxisAlignment.center,
                      children: [
                        Image.network(category['image'], height: 80),
                        Text(category['title']),
                      ],
                    ),
                  ),
                );
              },
            ),
    );
  }
}

class VideoPlayerScreen extends StatefulWidget {
  final List videos;
  final int initialIndex;
  final String title;
  final DateTime createdAt;
  const VideoPlayerScreen(
      {super.key,
      required this.videos,
      required this.initialIndex,
      required this.title,
      required this.createdAt});

  @override
  _VideoPlayerScreenState createState() => _VideoPlayerScreenState();
}

class _VideoPlayerScreenState extends State<VideoPlayerScreen> {
  late PageController _pageController;
  VideoPlayerController? _videoController;
  int currentIndex = 0;
  bool _isPlaying = false;
  @override
  void initState() {
    super.initState();
    currentIndex = widget.initialIndex;
    _pageController = PageController(initialPage: currentIndex);
    _initializeAndPlay(currentIndex);
  }

  Future<void> _initializeAndPlay(int index) async {
    try {
      _videoController?.dispose();

      _videoController =
          VideoPlayerController.network(widget.videos[index]['videos']);

      _videoController!.addListener(() {
        if (_videoController!.value.hasError) {
          print(
              "Video player error: ${_videoController!.value.errorDescription}");
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(content: Text("Error loading video")),
          );
        }
      });

      await _videoController!.initialize();
      setState(() {});
      _videoController!.play();
    } catch (e) {
      print("Failed to load video: $e");
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text("Error loading video: $e")),
      );
    }
  }

  @override
  void dispose() {
    _videoController?.dispose();
    _pageController.dispose();
    super.dispose();
  }

  void _onPageChanged(int index) {
    setState(() {
      currentIndex = index;
    });
    _initializeAndPlay(index);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: PageView.builder(
        controller: _pageController,
        onPageChanged: _onPageChanged,
        scrollDirection: Axis.vertical,
        itemCount: widget.videos.length,
        itemBuilder: (context, index) {
          return Stack(children: [
            Center(
              child: _videoController != null &&
                      _videoController!.value.isInitialized
                  ? AspectRatio(
                      aspectRatio: _videoController!.value.aspectRatio,
                      child: VideoPlayer(_videoController!),
                    )
                  : const Center(child: CircularProgressIndicator()),
            ),
            Positioned(
              top: 40.0,
              left: 16.0,
              right: 16.0,
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    widget.title,
                    style: TextStyle(
                        color: Colors.black,
                        fontSize: 20.0,
                        fontWeight: FontWeight.bold),
                  ),
                  Text(
                    timeago.format(widget.createdAt),
                    style: TextStyle(color: Colors.black, fontSize: 16.0),
                  ),
                ],
              ),
            ),
            // Add Play/Pause button on top of the video player
            Center(
              child: IconButton(
                icon: Icon(_isPlaying ? Icons.pause : Icons.play_arrow,
                    color: Colors.white, size: 50.0),
                onPressed: () {
                  setState(() {
                    _isPlaying
                        ? _videoController?.pause()
                        : _videoController?.play();
                    _isPlaying = !_isPlaying;
                  });
                },
              ),
            ),
          ]);
        },
      ),
    );
  }
}

import 'package:flutter/material.dart';
import 'package:park_app/models/Video.dart';
import 'package:video_player/video_player.dart';
import 'package:timeago/timeago.dart' as timeago;

class VideoPlayerScreen extends StatefulWidget {
  final List<Video> videos;
  final int initialIndex;
  final DateTime createdAt;
  final String title;
  const VideoPlayerScreen(
      {super.key,
      required this.videos,
      required this.initialIndex,
      required this.createdAt,
      required this.title});

  @override
  _VideoPlayerScreenState createState() => _VideoPlayerScreenState();
}

class _VideoPlayerScreenState extends State<VideoPlayerScreen> {
  late VideoPlayerController _controller;
  late int currentIndex;

  @override
  void initState() {
    super.initState();
    currentIndex = widget.initialIndex;
    _initializePlayer();
  }

  void _initializePlayer() {
    _controller =
        VideoPlayerController.network(widget.videos[currentIndex].videoUrl)
          ..initialize().then((_) {
            setState(() {});
            _controller.play();
          });
  }

  void _onVerticalDragEnd(DragEndDetails details) {
    if (details.primaryVelocity! < 0) {
      if (currentIndex < widget.videos.length - 1) {
        currentIndex++;
        _controller.dispose();
        _initializePlayer();
      }
    } else if (details.primaryVelocity! > 0) {
      if (currentIndex > 0) {
        currentIndex--;
        _controller.dispose();
        _initializePlayer();
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    final video = widget.videos[currentIndex];
    return GestureDetector(
      onVerticalDragEnd: _onVerticalDragEnd,
      onTap: () {
        setState(() {
          _controller.value.isPlaying
              ? _controller.pause()
              : _controller.play();
        });
      },
      child: Scaffold(
        backgroundColor: Colors.black,
        body: Stack(
          alignment: Alignment.center,
          children: [
            if (_controller.value.isInitialized)
              AspectRatio(
                aspectRatio: _controller.value.aspectRatio,
                child: VideoPlayer(_controller),
              ),
            Positioned(
              bottom: 50,
              left: 20,
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    video.title,
                    style: const TextStyle(color: Colors.white, fontSize: 20),
                  ),
                  Text(
                    timeago.format(
                        video.createdAt), // Assuming `timeAgo` is implemented
                    style: const TextStyle(color: Colors.white, fontSize: 14),
                  ),
                  VideoProgressIndicator(
                    _controller,
                    allowScrubbing: true,
                    colors: const VideoProgressColors(playedColor: Colors.red),
                  ),
                ],
              ),
            ),
            if (!_controller.value.isPlaying)
              const Icon(Icons.play_arrow, size: 80, color: Colors.white),
          ],
        ),
      ),
    );
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }
}
